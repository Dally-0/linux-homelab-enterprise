\section{Análisis Pormenorizado de las Vulnerabilidades}

\subsection{VULN-01: Intercambio Criptográfico Anónimo Diffie-Hellman en SMTP (25, 587)}
\begin{itemize}
  \item \textbf{Descripción Técnica:} En los puertos de correo \texttt{25/tcp} (SMTP) y \texttt{587/tcp} (Submission), el script NSE \texttt{ssl-dh-params} identificó la vulnerabilidad crítica de intercambio de llaves anónimo: \texttt{TLS\_DH\_anon\_WITH\_AES\_128\_GCM\_SHA256} con llaves de 1024 bits.
  \item \textbf{Cálculo de Severidad (CVSS v3.1):}
    \begin{itemize}
      \item \textbf{Vector CVSS:} \texttt{CVSS:3.1/AV:N/AC:H/PR:N/UI:N/S:U/C:H/I:H/A:N}
      \item \textbf{Puntaje Base:} \textbf{7.4 (Alta)}
    \end{itemize}
  \item \textbf{Análisis de Riesgo para la Organización:} El intercambio Diffie-Hellman anónimo (\texttt{DH\_anon}) encripta la comunicación frente a espionaje pasivo, pero \textbf{no autentica la identidad del servidor}. Un atacante situado en la misma red local puede realizar un ataque activo de intermediario (\textit{Man-in-the-Middle}), interceptar el handshake TLS, descifrar los correos bancarios y reenviarlos sin que los clientes ni los servidores perciban la intrusión.
\end{itemize}

\subsection{VULN-02: Exposición Pública del Puerto PostgreSQL (5432/tcp)}
\begin{itemize}
  \item \textbf{Descripción Técnica:} El servicio de base de datos relacional PostgreSQL versión 16.2 se encuentra accesible en la interfaz de red externa, permitiendo sondeos directos desde cualquier equipo de la red.
  \item \textbf{Cálculo de Severidad (CVSS v3.1):}
    \begin{itemize}
      \item \textbf{Vector CVSS:} \texttt{CVSS:3.1/AV:N/AC:L/PR:N/UI:N/S:U/C:H/I:N/A:N}
      \item \textbf{Puntaje Base:} \textbf{7.5 (Alta)}
    \end{itemize}
  \item \textbf{Análisis de Riesgo para la Organización:} Exponer el motor de datos a la LAN facilita ataques de fuerza bruta continuos contra las credenciales del usuario de base de datos y la explotación de vulnerabilidades en protocolos de transporte.
\end{itemize}

\subsection{VULN-03: Modo Debug Activo y Divulgación en Webmail (8080/tcp)}
\begin{itemize}
  \item \textbf{Descripción Técnica:} El script \texttt{http-aspnet-debug} reportó \texttt{status: DEBUG is enabled} en el puerto 8080 (Roundcube Webmail), indicando que el servidor web responde afirmativamente a verbos de depuración.
  \item \textbf{Cálculo de Severidad (CVSS v3.1):}
    \begin{itemize}
      \item \textbf{Vector CVSS:} \texttt{CVSS:3.1/AV:N/AC:L/PR:N/UI:N/S:U/C:L/I:N/A:N}
      \item \textbf{Puntaje Base:} \textbf{5.3 (Media)}
    \end{itemize}
  \item \textbf{Análisis de Riesgo para la Organización:} Los métodos de depuración activos pueden filtrar trazas de error de PHP/Apache, nombres de variables de sesión y rutas internas del servidor web, facilitando la construcción de ataques dirigidos.
\end{itemize}

\subsection{VULN-04: Transmisión de Sesiones y Credenciales sin Cifrado TLS (HTTP en Claro)}
\begin{itemize}
  \item \textbf{Descripción Técnica:} Servicios como el Webmail (8080), Filemanager (8082) y Grafana (3000) exponen interfaces web HTTP no cifradas en puertos alternativos.
  \item \textbf{Cálculo de Severidad (CVSS v3.1):}
    \begin{itemize}
      \item \textbf{Vector CVSS:} \texttt{CVSS:3.1/AV:N/AC:H/PR:N/UI:R/S:U/C:H/I:H/A:N}
      \item \textbf{Puntaje Base:} \textbf{7.4 (Alta)}
    \end{itemize}
  \item \textbf{Análisis de Riesgo para la Organización:} Captura pasiva de contraseñas de correo y cookies de sesión bancaria mediante envenenamiento ARP en la red conmutada.
\end{itemize}

\subsection{VULN-05: Ausencia de Cabeceras HTTP de Endurecimiento y Divulgación de Banners}
\begin{itemize}
  \item \textbf{Descripción Técnica:} Ausencia de cabeceras \texttt{X-Frame-Options}, \texttt{X-Content-Type-Options} y \texttt{HSTS} en peticiones directas.
  \item \textbf{Cálculo de Severidad (CVSS v3.1):}
    \begin{itemize}
      \item \textbf{Vector CVSS:} \texttt{CVSS:3.1/AV:N/AC:L/PR:N/UI:R/S:U/C:L/I:L/A:N}
      \item \textbf{Puntaje Base:} \textbf{5.3 (Media)}
    \end{itemize}
  \item \textbf{Análisis de Riesgo para la Organización:} Exposición a ataques de Clickjacking y MIME-sniffing en navegadores de clientes bancarios.
\end{itemize}

\subsection{VULN-06: Descarte Activo y Timeouts en Scripts NSE (\textit{Tarpitting Defensivo})}
\begin{itemize}
  \item \textbf{Descripción Técnica:} En los puertos POP3 (110), IMAP (143), IMAPS (993), POP3S (995) y en el analizador de inyección SQL (8080), Nmap reportó \texttt{TIMEOUT} y \texttt{ERROR: Script execution failed}.
  \item \textbf{Evaluación Defensiva:} Este hallazgo confirma la intercepción activa del sistema de defensa: el cortafuegos \texttt{nftables} y el motor \textbf{Suricata IPS} descartan las ráfagas de paquetes maliciosos, impidiendo que los scripts de ataque completen su ciclo de explotación.
\end{itemize}
